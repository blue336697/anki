'class Solution {'
'    //该题最简单的就是记忆化搜索（备忘录数组+dfs）'
'    int[][] visited;'
'    int n, m;'
'    public int longestIncreasingPath(int[][] matrix) {'
'        int maxRes = 0;'
'        n = matrix.length;'
'        m = matrix[0].length;'
'        visited = new int[n][m];'
'        for(int i = 0; i &lt; n; i++){'
'            for(int j = 0; j &lt; m; j++){'
'                maxRes = Math.max(maxRes, dfs(matrix, i, j, Integer.MIN_VALUE));'
'            }'
'        }'
'        return maxRes;'
'    }'
''
'    public int dfs(int[][] matrix, int i, int j, int preTemp){'
'        //边界判断'
'        if(i &gt;= n || j &gt;= m || i &lt;0 || j &lt; 0)'
'            return 0;'
'        if(matrix[i][j] &lt;= preTemp)'
'            return 0;'
'        //如果之前已经计算过直接返回'
'        if(visited[i][j] &gt; 0)'
'            return visited[i][j];'
'        //往下'
'        int under = dfs(matrix, i + 1, j, matrix[i][j]);'
'        //往上'
'        int top = dfs(matrix, i - 1, j, matrix[i][j]);'
'        //往左'
'        int left = dfs(matrix, i, j - 1, matrix[i][j]);'
'        //往右'
'        int right = dfs(matrix, i, j + 1, matrix[i][j]);'
''
'        visited[i][j] = Math.max(Math.max(under, top), Math.max(left, right)) + 1;'
'        return visited[i][j];'
'    }'
'}'

'        if(len == 0)'
'            return 0;'
'        int[] dp = new int[len];'
''
'        //至少发一个糖果'

'        for(int nums : dp)'
'            res += nums;'

'            dp[0][m] = 0; // zero egg'

'            return heights[0];'
''
'        int[] newHeights = new int[len + 2];'
'        newHeights[0] = 0;'

'        //转成double计算，丢坑里了'
'        //[8,1,6,6] 6÷(1-6÷8) =6÷(1-6/8) =6÷2/8 =6x8/2 =6x4 =24'

'    }'
''
'    public boolean judgePoint24(double[] nums) {'
'        //递归出口，剩余一个数，判断是不是24点'
'        if (nums.length == 1) {'
'            //处理计算精度问题'
'            return nums[0] &gt; 23.999 &amp;&amp; nums[0] &lt; 24.001;'
'        }'
''
'        //任意拿两个不相同的数，通过运算符组成第三个数，进行后续的24点计算'
'        //其中，括号不用考虑，因为是任意两个数的所有允许符都参与，所以有括号，没括号的情况，都包含在内了'
'        for (int x = 0; x &lt; nums.length - 1; x++) {'
'            for (int y = x + 1; y &lt; nums.length; y++) {'
'                //是否合法'
'                boolean isValid = false;'
''
'                //每次两数操作 出来的数组是原来数据的length-1'
'                //删除后面的元素（这样不会影响前面的元素），前面的元素用来放置新值'
'                //其实就是将y对应的那个数在每一层的temp中删除，y与x运算后的结果就放在x的位置上'
'                //这样每一次就会缩减数组'
'                double[] temp = new double[nums.length - 1];'
'                //copy待删除元素y的前部'
'                System.arraycopy(nums, 0, temp, 0, y);'
'                //copy待删除元素y的后部'
'                System.arraycopy(nums, y + 1, temp, y, temp.length - y);'
''
'                //只要下面某一个返回了true，直接会跳过其他的运算符，后面的也不会算了'
''
'                //加法'
'                temp[x] = nums[x] + nums[y];'
'                isValid = isValid || judgePoint24(temp);'
''
'                //减法（减与被减）'
'                temp[x] = nums[x] - nums[y];'
'                isValid = isValid || judgePoint24(temp);'
'                temp[x] = nums[y] - nums[x];'
'                isValid = isValid || judgePoint24(temp);'
''
'                //乘法'
'                temp[x] = nums[x] * nums[y];'
'                isValid = isValid || judgePoint24(temp);'
''
'                //除法，除与被除'
'                if (nums[y] != 0) {'
'                    temp[x] = nums[x] / nums[y];'
'                    isValid = isValid || judgePoint24(temp);'
'                }'
'                if (nums[x] != 0) {'
'                    temp[x] = nums[y] / nums[x];'
'                    isValid = isValid || judgePoint24(temp);'
'                }'
''
'                //合法就返回'
'                if (isValid) {'
'                    return true;'
'                }'
'            }'
'        }'

'class Solution {'
'    public int largestRectangleArea(int[] heights) {'
'        int len = heights.length;'
'        if (len == 0) {'
'            return 0;'
'        }'
'        if (len == 1) {'
'            return heights[0];'
'        }'
''
'        //开始缔造单调栈，这个栈中进栈确定每一个圆柱体的宽度，每个进栈的圆柱体如果遇到比自己矮'
'        //的圆柱体就可以确定了，可以将自己出栈表示确定完成'
'        //例如 2 1 5 6 2 3，这里需要注意当遍历到3的时候，不存在小于3的元素了，这时候就要将它的'
'        //右边看做一个高度为0，下标为6的圆柱体，这就引出了如何优化当前，即哨兵机制'
'        int res = 0;'
'        Deque&lt;Integer&gt; stack = new ArrayDeque&lt;&gt;(len);'
'        for (int i = 0; i &lt; len; i++) {'
'            // 这个 while 很关键，因为有可能不止一个柱形的最大宽度可以被计算出来'
'            while (!stack.isEmpty() &amp;&amp; heights[i] &lt; heights[stack.peekLast()]) {'
'                //遍历到i=1时进入此循环'
'                int curHeight = heights[stack.pollLast()];  //获取前面高的圆柱体即为2'
'                //如果弹出栈的跟栈顶元素相等，则弹出栈顶元素 '
'                while (!stack.isEmpty() &amp;&amp; heights[stack.peekLast()] == curHeight) {'
'                    stack.pollLast();'
'                }'
'                //当前宽度'
'                int curWidth;'
'                if (stack.isEmpty()) {  //栈为空说明都确定完宽了，当前元素的左边界就是自身'
'                    curWidth = i;'
'                } else {    //否则就是当前的下标-栈顶的下标，如上面例子碰到第二个2时，会将6弹出'
'                //它的宽度就是 4(右边界) - 2(左边界) - 1，减2是因为将6弹出了，'
'                //现在栈顶就是5即下标为2，你可以根据图看到，左右边界都是严格小于前面高的圆柱体'
'                    curWidth = i - stack.peekLast() - 1;'
'                }'
'                //然后记录'
'                res = Math.max(res, curHeight * curWidth);'
'            }'
'            stack.addLast(i);'
'        }'
'        //最后数组遍历完如果还有没有确定宽度的则进行处理'
'        while (!stack.isEmpty()) {'
'            int curHeight = heights[stack.pollLast()];'
'            while (!stack.isEmpty() &amp;&amp; heights[stack.peekLast()] == curHeight) {'
'                stack.pollLast();'
'            }'
'            int curWidth;'
'            if (stack.isEmpty()) {'
'                curWidth = len;'
'            } else {'
'                curWidth = len - stack.peekLast() - 1;'
'            }'
'            res = Math.max(res, curHeight * curWidth);'
'        }'
'        return res;'
'    }'
'}'

'public class Solution {'
'    public int splitArray(int[] nums, int m) {'
'        int max = 0;'
'        int sum = 0;'
''
'        // 计算「子数组各自的和的最大值」的上下界'
'        for (int num : nums) {'
'            max = Math.max(max, num);'
'            sum += num;'
'        }'
''
'        // 使用「二分查找」确定一个恰当的「子数组各自的和的最大值」，'
'        // 使得它对应的「子数组的分割数」恰好等于 m'
'        int left = max;'
'        int right = sum;'
'        while (left &lt; right) {'
'            int mid = left + (right - left) / 2;'
''
'            int splits = split(nums, mid);'
'            if (splits &gt; m) {'
'                // 如果分割数太多，说明「子数组各自的和的最大值」太小，此时需要将「子数组各自的和的最大值」调大'
'                // 下一轮搜索的区间是 [mid + 1, right]'
'                left = mid + 1;'
'            } else {'
'                // 下一轮搜索的区间是上一轮的反面区间 [left, mid]'
'                right = mid;'
'            }'
'        }'
'        return left;'
'    }'
''
'    /***'
'     *'
'     * @param nums 原始数组'
'     * @param maxIntervalSum 子数组各自的和的最大值'
'     * @return 满足不超过「子数组各自的和的最大值」的分割数'
'     */'
'    private int split(int[] nums, int maxIntervalSum) {'
'        // 至少是一个分割'
'        int splits = 1;'
'        // 当前区间的和'
'        int curIntervalSum = 0;'
'        for (int num : nums) {'
'            // 尝试加上当前遍历的这个数，如果加上去超过了「子数组各自的和的最大值」，就不加这个数，另起炉灶'
'            if (curIntervalSum + num &gt; maxIntervalSum) {'
'                curIntervalSum = 0;'
'                splits++;'
'            }'
'            curIntervalSum += num;'
'        }'

'        int cardBuckets = 0, n = nums.length;'
'        int[] top = new int[n];'
'        for (int i = 0; i &lt; n; i++) {'
'            // 要处理的扑克牌'
'            int poker = nums[i];'
'            int left = 0, right = cardBuckets;'
'            // 二分查找插入位置'
'            while (left &lt; right) {'
'                int mid = (left + right) / 2;'
'                if (top[mid] &gt;= poker)'
'                    right = mid;'
'                else'
'                    left = mid + 1;'

'        }'
'        // 牌堆数就是 LIS 长度'

'    }'

'class Solution {'
'    public int minInsertions(String s) {'
'        int len = s.length();'
'        //注意这里定义状态为：dp[i][j]：i到j的最长回文子序列为多少'
'        int[][] dp = new int[len][len];'
'        //初始化，我们这样初始化以及遍历规则其实就是把二维的下半部分格子全部都填了'
'        for(int i = 0; i &lt; len; i++){'
'            dp[i][i] = 1;   //自己跟自己比回文肯定为1，i=j时就是i到i本身了呗'
'        }'
'        //从左到右，从下到上的遍历，因为我们格子的下半部分已知的已经填好了'

'            for(int j = i + 1; j &lt; len; j++){'
'                if(s.charAt(i) == s.charAt(j))'
'                    dp[i][j] = dp[i+1][j-1] + 2;'
'                else'
'                    dp[i][j] = Math.max(dp[i+1][j], dp[i][j-1]);'
'            }'
'        }'
'        //由于状态定义我们最后返回的也是整个字符串的最长回文子序列'
'        return len - dp[0][len - 1];'
'    }'

'class Solution {'
'    public int longestSubstring(String s, int k) {'
'        if(k == 0)'
'            return 0;'
'        Map&lt;Character, Integer&gt; map = new HashMap&lt;&gt;();'
'        int maxLen = 0, len = s.length();'
'        //窗口内字符的种类'
'        int count = 0;'
'        //滑动窗口'
'        for(int left = 0, right = 0; right &lt; len; right++){'
'            //开始移动右窗口，纳入新字符，记录字符在窗口内的出现次数'
'            map.put(s.charAt(right), map.getOrDefault(s.charAt(right), 0) + 1);'
'            //我们需要找到k个种类字符的子串，所以当前字符出现了，就纳入count'
'            if(map.get(s.charAt(right)) == 1)'
'                count++;'
'            //满足k的时候，就要开始收紧左边界，找出最长的子串'
'            while(count &gt; k){'
'                //收紧左边界，然后减去次数'
'                map.put(s.charAt(left), map.get(s.charAt(left)) - 1);'
'                //如果此时刚好为0，这个字符就出现过一次后，那么将count减一'
'                if(map.get(s.charAt(left)) == 0)'
'                    count--;'
'                //收紧边界'
'                left++;'
'            }'
'            //然后此时count == k，就看子串的长度谁长'
'            maxLen = Math.max(maxLen, right - left + 1);'
'        }'
'        return maxLen;'
'    }'
'}'
