```java
//利用深度递归解决，可以看图，并加记住这个模板，他可以解决岛屿中的问题，还有一题岛屿面积问题也是这个模板。
    public static int maxAreaOfIsland(int[][] grid) {
        //定义一个表示岛屿数量的变量
        int maxArea = 0;
        //这个两层for循环是用来遍历整张二维表格中所有的陆地
        //其中 i 表示行，j 表示列
        for(int i = 0; i < grid.length; i++){
            for(int j =0;j < grid[0].length; j++){
                //取出所有的陆地
                if(grid[i][j] == 1){
                    //深度递归，遍历所有的陆地
                    maxArea = Math.max(dfs(grid,i,j), maxArea);
                    //用来统计有多少岛屿，岛屿是由多个陆地组成的，概念不一样
                    //利用递归和回溯的效果，每次dfs(grid,i,j);跳出来执行count++
                    //正好就是一个岛
                    //count++;
                }
            }
        }
        //返回岛屿的数量
        return maxArea;
    }
    public static int dfs(int[][] grid,int i, int j){
        //防止 i 和 j 越界，也就是防止超出岛屿（上下左右）的范围。
        //特别注意当遍历到海洋的时候和已遍历到陆地也退出循环
        if(i < 0 || j < 0 || i >= grid.length ||
                j >= grid[0].length || grid[i][j] == 0 ||
                grid[i][j] == 2)
            return 0;

        //将遍历过的陆地改为海洋，防止重复遍历
        grid[i][j] = 2;
        return 1 + dfs(grid, i + 1, j) +
                dfs(grid, i - 1, j ) +
                dfs(grid, i, j + 1) +
                dfs(grid, i, j - 1);

    }
```

```java
public static int dfs(int[][] grid,int i, int j, int maxArea){
        //防止 i 和 j 越界，也就是防止超出岛屿（上下左右）的范围。
        //特别注意当遍历到海洋的时候和已遍历到陆地也退出循环
        if(i < 0 || j < 0 || i >= grid.length ||
                j >= grid[0].length || grid[i][j] == 0 ||
                grid[i][j] == 2)
            return 0;
        maxArea++;
        //将遍历过的陆地改为海洋，防止重复遍历
        grid[i][j] = 2;
        //下
        maxArea = Math.max(dfs(grid,i+1,j,maxArea), maxArea);
        //上
        maxArea = Math.max(dfs(grid,i-1,j,maxArea), maxArea);
        //右
        maxArea = Math.max(dfs(grid,i,j+1,maxArea), maxArea);
        //左
        maxArea = Math.max(dfs(grid,i,j-1,maxArea), maxArea);

        return maxArea;
    }
```

```java
class Solution {
    // 节点的入度: 使用数组保存每个节点的入度,
    public boolean canFinish(int numCourses, int[][] prerequisites) {
        // 1.课号和对应的入度
        Map<Integer, Integer> inDegree = new HashMap<>();
        // 将所有的课程先放入
        for (int i = 0; i < numCourses; i++) {
            inDegree.put(i, 0);
        }
        // 2.依赖关系, 依赖当前课程的后序课程
        Map<Integer, List<Integer>> adj = new HashMap<>();

        // 初始化入度和依赖关系
        for (int[] relate : prerequisites) {
            // (3,0), 想学3号课程要先完成0号课程, 更新3号课程的入度和0号课程的依赖(邻接表)
            int cur = relate[1];
            int next = relate[0];
            // 1.更新入度
            inDegree.put(next, inDegree.get(next) + 1);
            // 2.当前节点的邻接表
            if (!adj.containsKey(cur)) {
                adj.put(cur, new ArrayList<>());
            }
            adj.get(cur).add(next);
        }

        // 3.BFS, 将入度为0的课程放入队列, 队列中的课程就是没有先修, 可以学的课程
        Queue<Integer> q = new LinkedList<>();
        for (int key : inDegree.keySet()) {
            if (inDegree.get(key) == 0) {
                q.offer(key);
            }
        }
        int count = 0;
        // 取出一个节点, 对应学习这门课程.
        // 遍历当前邻接表, 更新其入度; 更新之后查看入度, 如果为0, 加入到队列
        while (!q.isEmpty()) {
            int cur = q.poll();
            count++;
            // 遍历当前课程的邻接表, 更新后继节点的入度
            if (!adj.containsKey(cur)) {
                continue;
            }
            List<Integer> successorList = adj.get(cur);

            for (int k : successorList) {
                inDegree.put(k, inDegree.get(k) - 1);
                if (inDegree.get(k) == 0) {
                    q.offer(k);
                }
            }
        }

        // 4.选了的课等于总课数，true，否则false
        return count == numCourses;

        /*
        // 4.遍历入队, 如果还有课程的入度不为0, 返回fasle
        for (int key : inDegree.keySet()) {
            if (inDegree.get(key) != 0) {
                return false;
            }
        }
        return true;
        */
    }
}
```

**拓扑排序+DFS**

```java
class Solution {
    //这道题是图的关系，我们的思路就找有无环，有环就返回false
    //通过常见邻接表来表示图的每个节点的连接关系
    public boolean canFinish(int numCourses, int[][] prerequisites) {
        //设置一个存储总共课程的集合
        List<List<Integer>> adjacency = new ArrayList<>();
        //总共有多少门课程，adjacency中就有多少个list
        //即有多少个节点就有多少个list
        for(int i = 0; i < numCourses; i++)
            adjacency.add(new ArrayList<>());
        //对应每个节点的访问记录，标记是否已经访问过
        //0：未被访问过 
        //-1：已被其他节点为起点的遍历访问 
        //1：已被当前节点为起点的遍历访问
        int[] flags = new int[numCourses];
        for(int[] cp : prerequisites)
            //将需要先学的课程的值得到，该值对应的邻接表的位置
            //就是该节点的位置
            //并将前置课程学习完后才能学习的课程加入到该位置
            //例如构成第0号位置的list中存放了1号课程
            //表达的意思就是学完0号在学1号 即[1,0]
            adjacency.get(cp[1]).add(cp[0]);
        for(int i = 0; i < numCourses; i++)
            //循环遍历以不同节点为起点的dfs 只要返回false，
            //if成立即说明有环，返回false
            if(!dfs(adjacency, flags, i)) return false;
        return true;
    }

    private boolean dfs(List<List<Integer>> adjacency, int[] flags, int i) {
        if(flags[i] == 1) return false;
        if(flags[i] == -1) return true;
        //如果未被访问过，则标记
        flags[i] = 1;
        //然后遍历对应位置的后续学习课程
        for(Integer j : adjacency.get(i))
            //这里就是循环遍历每个节点作为起点，例：[[1,0],[0,1]]
            //例如在list中0的位置存在是1,1的位置存放是0
            //先访问0，然后在flag的0位置标记为1，然后将list的0位置的值0=1取出
            //在进入dfs判断1为起点，将flag中1的位置标记为1，然后将list1
            //的位置中的0取出，再次进入判断，此时读取到的i为0，flag0的位置
            //为1，说明存在环
            if(!dfs(adjacency, flags, j)) return false;
        flags[i] = -1;
        return true;
    }
}
```

### 检测循环依赖

题目描述：https://mp.weixin.qq.com/s/pCRscwKqQdYYN7M1Sia7xA

https://www.cnblogs.com/bigsai/p/11489260.html

```java
public class Solution {
    public List<Integer> haveCircularDependency(int n, List<List<Integer>> prerequisites) {
        List<List<Integer>> g = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            g.add(new ArrayList<>());
        }
        int[] indeg = new int[n];
        List<Integer> res = new ArrayList<>();

        for (List<Integer> pre : prerequisites) {
            int a = pre.get(0), b = pre.get(1);
            g.get(a).add(b);
            indeg[b]++;
        }

        Queue<Integer> q = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            if (indeg[i] == 0) q.offer(i);
        }

        while (!q.isEmpty()) {
            int t = q.poll();
            res.add(t);
            for (int j : g.get(t)) {
                indeg[j]--;
                if (indeg[j] == 0) {
                    q.offer(j);
                }
            }
        }

        if (res.size() == n) return res;
        else return new ArrayList<>();
    }
}
```

```java
class Solution {
    public int[] findOrder(int numCourses, int[][] prerequisites) {
        // 1.课号和对应的入度
        Map<Integer, Integer> inDegree = new HashMap<>();
        // 将所有的课程先放入
        for (int i = 0; i < numCourses; i++) {
            inDegree.put(i, 0);
        }
        // 2.依赖关系, 依赖当前课程的后序课程
        Map<Integer, List<Integer>> adj = new HashMap<>();

        // 初始化入度和依赖关系
        //入度：指明有几个前置任务  依赖关系：指明前置任务具体是啥
        for (int[] relate : prerequisites) {
            // (3,0), 想学3号课程要先完成0号课程, 更新3号课程的入度和0号课程的依赖(邻接表)
            int cur = relate[1];
            int next = relate[0];
            // 1.更新入度
            inDegree.put(next, inDegree.get(next) + 1);
            // 2.当前节点的邻接表
            if (!adj.containsKey(cur)) {
                adj.put(cur, new ArrayList<>());
            }
            //更新依赖关系，即上3需要先完成0,0——————》3，即3的入度为1，这个1代表课程0
            adj.get(cur).add(next);
        }

        // 3.BFS, 将入度为0的课程放入队列, 队列中的课程就是没有先修, 可以学的课程
        Queue<Integer> q = new LinkedList<>();
        for (int key : inDegree.keySet()) {
            if (inDegree.get(key) == 0) {
                q.offer(key);
            }
        }

        int[] res = new int[numCourses];
        int i = 0;
        // 取出一个节点, 对应学习这门课程.
        // 遍历当前邻接表, 更新其入度; 更新之后查看入度, 如果为0, 加入到队列
        while (!q.isEmpty()) {
            //从这里取出来的节点入度都为0
            int cur = q.poll();
            //在这就要更新，因为最后一个没有任何依赖关系的节点是不存在与邻接表中的，他就是结果集
            //的最后一个节点，所以要在这就加入进去
            res[i++] = cur;
            // 遍历当前课程的邻接表, 更新后继节点的入度
            //入度为0，但又不存在邻接表中说明没有依赖关系即没有后续了
            if (!adj.containsKey(cur)) {        
                continue;
            }
            List<Integer> successorList = adj.get(cur);
            //更新入度列表，将新的入度更新进去，并将入度为0的加入队列
            for (int k : successorList) {
                inDegree.put(k, inDegree.get(k) - 1);
                if (inDegree.get(k) == 0) {
                    q.offer(k);
                }
            }
        }
        //如果存在回路关系，我们遍历当进行消除入度后，如果还存在入度不为0的就说明存在环
        for (int key : inDegree.keySet()) {
            if (inDegree.get(key) != 0) {
                return new int[0];
            }
        }
        return res;
    }
}
```
